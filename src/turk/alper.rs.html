318fn kavram_makale(konu: &str) -> Option<&'static str> {
319    let n = normalize_tr(konu);
320    if n.contains("relu") {
321        Some(
322            "**ReLU**\n- `f(x)=max(0,x)`\n- Vanishing gradient’i azaltır\n- TÜRK: `yogun(..., \"relu\")`",
323        )
324    } else if n.contains("softmax") {
325        Some("**Softmax**\n- Çok sınıflı olasılık; toplam 1\n- Cross-entropy ile çift\n- ALPER son katmanda Softmax kullanır")
326    } else if n.contains("adam") {
327        Some("**Adam**\n- Momentum + RMSprop\n- XOR `lr≈0.05`, ALPER `lr≈0.035`\n- `derle(adam, ...)`")
328    } else if n.contains("dropout") {
329        Some("**Dropout**\n- Eğitimde nöron kapatma → genelleme\n- `katman_ekle(dropout(0.1))`")
330    } else if n.contains("wasm") {
331        Some("**WASM**\n- Tarayıcıda native’e yakın baytkod\n- Kokpit/playground aynı paketi paylaşır")
332    } else if n.contains("overfit") || n.contains("asiri uyum") || n.contains("aşırı") {
333        Some("**Overfitting**\n- Train iyi, val kötü\n- Dropout, early stopping, daha fazla veri")
334    } else if n.contains("mse") {
335        Some("**MSE**\n- `(ŷ-y)²` ortalaması\n- XOR/regresyon demolarında klasik")
336    } else if n.contains("cross") || n.contains("entropy") {
337        Some("**Cross-Entropy**\n- Sınıflandırma loss’u\n- Softmax ile birlikte")
338    } else if n.contains("mlp") || n.contains("sinir") {
339        Some("**MLP**\n- Dense yığını\n- ALPER: girdi→64→32→niyet (softmax)")
340    } else if n.contains("backprop") || n.contains("geri yay") {
341        Some("**Backpropagation**\n- Zincir kuralı ile `∂L/∂W`\n- Forward → loss → backward → update")
342    } else if n.contains("ogrenme") || n.contains("learning") || n.contains("lr") {
343        Some("**Öğrenme oranı**\n- Adım büyüklüğü η\n- Yüksek → salınım; düşük → yavaş yakınsama")
344    } else if n.contains("tf") || n.contains("idf") || n.contains("bigram") {
345        Some("**TF-IDF + Bigram**\n- `benim_adim` vs `senin_adin`\n- ALPER giriş vektörü")
346    } else if n.contains("embed") || n.contains("gomme") || n.contains("gömme") {
347        Some("**Embedding**\n- `gommeli(kelime=..., boyut=...)`\n- Lookup tablosu; TF-IDF’siz kelime vektörleri")
348    } else if n.contains("conv1") || n.contains("evrisim") {
349        Some("**Conv1D**\n- `evrisim1d(giris=..., cikis=..., cekirdek=...)`\n- 1B kayan pencere + geri yayılım")
350    } else {
351        None
352    }
353}

379static STOP: &[&str] = &[
380    "ve", "veya", "ile", "bir", "bu", "su", "o", "da", "de", "ki", "mi", "mu", "icin", "gibi",
381    "kadar", "daha", "en", "cok", "ya", "ama", "var", "yok",
382];

384pub fn normalize_tr(s: &str) -> String {
385    let lower = s.to_lowercase();
386    let mut out = String::with_capacity(lower.len());
387    for c in lower.chars() {
388        let mapped = match c {
389            'ı' | 'İ' => 'i',
390            'ğ' | 'Ğ' => 'g',
391            'ü' | 'Ü' => 'u',
392            'ş' | 'Ş' => 's',
393            'ö' | 'Ö' => 'o',
394            'ç' | 'Ç' => 'c',
395            _ => c,
396        };
397        if mapped.is_ascii_alphanumeric() || mapped == ' ' || mapped == '_' || mapped == '-' {
398            out.push(mapped);
399        } else if mapped.is_whitespace() {
400            out.push(' ');
401        }
402    }
403    out.split_whitespace().collect::<Vec<_>>().join(" ")
404}

406fn stem_tr(tok: &str) -> String {
407    let suffixes = [
408        "lariniz", "leriniz", "iyorlar", "uyorlar", "makta", "mekte", "lardan", "lerden", "iyor",
409        "uyor", "arak", "erek", "dir", "dur", "tir", "tur", "dan", "den", "nin", "nun", "lar",
410        "ler", "lik", "luk", "mak", "mek", "in", "un", "im", "um",
411    ];
412    let mut t = tok.to_string();
413    for s in suffixes {
414        if t.len() > s.len() + 2 && t.ends_with(s) {
415            t.truncate(t.len() - s.len());
416            break;
417        }
418    }
419    t
420}

422pub fn tokenize(text: &str) -> Vec<String> {
423    let n = normalize_tr(text);
424    let uni: Vec<String> = n
425        .split_whitespace()
426        .map(stem_tr)
427        .filter(|t| t.len() > 1 && !STOP.contains(&t.as_str()))
428        .collect();
429    let mut out = uni.clone();
430    for i in 0..uni.len().saturating_sub(1) {
431        out.push(format!("{}_{}", uni[i], uni[i + 1]));
432    }
433    out
434}

479impl AlperBeyin {
480    /// Önbellekten yükler veya eğitir.
481    pub fn ac() -> Result<Self> {
482        let mut profile = UserProfile::default();
483        if let Ok(raw) = fs::read_to_string(profile_path()) {
484            if let Ok(p) = serde_json::from_str::<UserProfile>(&raw) {
485                profile = p;
486            }
487        }
488
489        if let Ok(beyin) = Self::onbellekten_yukle(profile.clone()) {
490            return Ok(beyin);
491        }
492
493        let mut beyin = Self::bos(profile)?;
494        beyin.egit_ve_kaydet()?;
495        Ok(beyin)
496    }
497
498    fn bos(profile: UserProfile) -> Result<Self> {
499        Ok(Self {
500            model: Sequential::new(),
501            vocab: Vec::new(),
502            profile,
503            last_intent: None,
504            last_topic: None,
505            accuracy: 0.0,
506            sample_count: 0,
507            from_cache: false,
508        })
509    }
510
511    fn fingerprint_ok(meta: &BrainCacheMeta) -> bool {
512        meta.schema == CACHE_SCHEMA
513            && meta.version == ALPER_VERSION
514            && meta.intents.len() == INTENTS.len()
515            && meta.intents.iter().zip(INTENTS.iter()).all(|(a, b)| a == b)
516    }
517
518    fn onbellekten_yukle(profile: UserProfile) -> Result<Self> {
519        let meta_raw = fs::read_to_string(meta_path())
520            .map_err(|e| TurkError::io_error(&format!("ALPER meta okunamadı: {e}")))?;
521        let meta: BrainCacheMeta = serde_json::from_str(&meta_raw)
522            .map_err(|e| TurkError::deserialization_error(&format!("ALPER meta: {e}")))?;
523        if !Self::fingerprint_ok(&meta) {
524            return Err(TurkError::RuntimeError(
525                "ALPER önbellek sürümü uyuşmuyor".into(),
526            ));
527        }
528        let mut model = Sequential::new();
529        model.modeli_ikili_yukle(brain_bin_path().to_str().unwrap_or("alper_brain.bin"))?;
530        // Softmax sınıflandırma için CE bağla
531        let lr: Float = 0.035;
532        model.compile_with_loss(
533            Box::new(Adam::new(lr, 0.9, 0.999, 1e-8)),
534            Box::new(CrossEntropy::new(false)),
535        );
536        Ok(Self {
537            model,
538            vocab: meta.vocab,
539            profile,
540            last_intent: None,
541            last_topic: None,
542            accuracy: meta.accuracy,
543            sample_count: meta.sample_count,
544            from_cache: true,
545        })
546    }
547
548    fn build_vocab(docs: &[String]) -> Vec<VocabTerm> {
549        let mut df: HashMap<String, usize> = HashMap::new();
550        for doc in docs {
551            let uniq: HashSet<String> = tokenize(doc).into_iter().collect();
552            for t in uniq {
553                *df.entry(t).or_default() += 1;
554            }
555        }
556        let n = docs.len() as f64;
557        let mut terms: Vec<_> = df.into_iter().collect();
558        terms.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
559        terms.truncate(VOCAB_LIMIT);
560        terms
561            .into_iter()
562            .map(|(term, d)| VocabTerm {
563                idf: ((1.0 + n) / (1.0 + d as f64)).ln() + 1.0,
564                term,
565            })
566            .collect()
567    }
568
569    fn vectorize(&self, text: &str) -> Vec<Float> {
570        let toks = tokenize(text);
571        let mut tf: HashMap<&str, f64> = HashMap::new();
572        for t in &toks {
573            *tf.entry(t.as_str()).or_default() += 1.0;
574        }
575        let len = toks.len().max(1) as f64;
576        self.vocab
577            .iter()
578            .map(|v| {
579                let c = tf.get(v.term.as_str()).copied().unwrap_or(0.0);
580                (c / len * v.idf) as Float
581            })
582            .collect()
583    }
584
585    fn training_matrices(&self) -> Result<(Array2<Float>, Array2<Float>, usize)> {
586        let ds = dataset();
587        let mut phrases: Vec<(usize, String)> = Vec::new();
588        for (i, intent) in INTENTS.iter().enumerate() {
589            if let Some(list) = ds.get(intent) {
590                for p in list {
591                    phrases.push((i, (*p).to_string()));
592                }
593            }
594        }
595        let n = phrases.len();
596        let d = self.vocab.len();
597        let c = INTENTS.len();
598        let mut x = Array2::zeros((n, d));
599        let mut y = Array2::zeros((n, c));
600        for (row, (idx, phrase)) in phrases.iter().enumerate() {
601            let v = {
602                // geçici vocab ile vektörle
603                let toks = tokenize(phrase);
604                let mut tf: HashMap<&str, f64> = HashMap::new();
605                for t in &toks {
606                    *tf.entry(t.as_str()).or_default() += 1.0;
607                }
608                let len = toks.len().max(1) as f64;
609                self.vocab
610                    .iter()
611                    .map(|vt| {
612                        let c = tf.get(vt.term.as_str()).copied().unwrap_or(0.0);
613                        (c / len * vt.idf) as Float
614                    })
615                    .collect::<Vec<_>>()
616            };
617            for (j, val) in v.into_iter().enumerate() {
618                x[[row, j]] = val;
619            }
620            y[[row, *idx]] = 1.0;
621        }
622        Ok((x, y, n))
623    }

625    /// Zorla yeniden eğit ve diske yaz.
626    pub fn egit_ve_kaydet(&mut self) -> Result<f64> {
627        let ds = dataset();
628        let docs: Vec<String> = INTENTS
629            .iter()
630            .flat_map(|i| {
631                ds.get(i)
632                    .into_iter()
633                    .flat_map(|v| v.iter().map(|s| (*s).to_string()))
634            })
635            .collect();
636        self.vocab = Self::build_vocab(&docs);
637        let (x, y, n) = self.training_matrices()?;
638        let input_dim = self.vocab.len();
639        let output_dim = INTENTS.len();
640
641        let mut model = Sequential::new();
642        model
643            .add(dense(input_dim, HIDDEN1, Activation::Relu)?)
644            .add(dense(HIDDEN1, HIDDEN2, Activation::Relu)?)
645            .add(dense(HIDDEN2, output_dim, Activation::Softmax)?);
646        let lr: Float = 0.035;
647        model.compile_with_loss(
648            Box::new(Adam::new(lr, 0.9, 0.999, 1e-8)),
649            Box::new(CrossEntropy::new(false)),
650        );
651        let _hist = model.fit(&x.view(), &y.view(), TRAIN_EPOCHS, TRAIN_BATCH.min(n), None)?;
652
653        // Doğruluk
654        let pred = model.predict(&x.view())?;
655        let mut correct = 0usize;
656        for i in 0..n {
657            let mut best = 0usize;
658            let mut best_v = pred[[i, 0]];
659            for j in 1..output_dim {
660                if pred[[i, j]] > best_v {
661                    best_v = pred[[i, j]];
662                    best = j;
663                }
664            }
665            let mut true_j = 0usize;
666            for j in 0..output_dim {
667                if y[[i, j]] > 0.5 {
668                    true_j = j;
669                    break;
670                }
671            }
672            if best == true_j {
673                correct += 1;
674            }
675        }
676        let acc = correct as f64 / n as f64;
677        self.model = model;
678        self.accuracy = acc;
679        self.sample_count = n;
680        self.from_cache = false;
681        self.kaydet_onbellek()?;
682        Ok(acc)
683    }

722    pub fn siniflandir(&mut self, text: &str) -> Result<IntentSonuc> {
723        if self.vocab.is_empty() || self.model.layers.is_empty() {
724            return Err(TurkError::ModelError("ALPER beyni hazır değil".into()));
725        }
726        let v = self.vectorize(text);
727        let x = Array2::from_shape_vec((1, v.len()), v)
728            .map_err(|e| TurkError::ShapeMismatch(e.to_string()))?;
729        let pred = self.model.predict(&x.view())?;
730        let mut best = 0usize;
731        let mut best_v = pred[[0, 0]];
732        for j in 1..INTENTS.len() {
733            if pred[[0, j]] > best_v {
734                best_v = pred[[0, j]];
735                best = j;
736            }
737        }
738        Ok(IntentSonuc {
739            intent: INTENTS[best].to_string(),
740            confidence: best_v,
741            index: best,
742        })
743    }

745    fn extract_name(text: &str) -> Option<String> {
746        let lower = text.to_lowercase();
747        for key in ["benim adım ", "benim adim ", "adım ", "adim ", "ismim "] {
748            if let Some(rest) = lower.split(key).nth(1) {
749                let name = rest
750                    .split_whitespace()
751                    .next()?
752                    .trim_matches(|c: char| !c.is_alphabetic())
753                    .to_string();
754                if name.len() >= 2 {
755                    let mut chars = name.chars();
756                    let first = chars.next()?.to_uppercase().to_string();
757                    return Some(first + chars.as_str());
758                }
759            }
760        }
761        None
762    }

764    fn safe_eval(text: &str) -> Option<f64> {
765        // Operatörleri koru — normalize_tr +/* siler
766        let mut expr = text.to_lowercase();
767        for (a, b) in [
768            ('ı', 'i'),
769            ('ğ', 'g'),
770            ('ü', 'u'),
771            ('ş', 's'),
772            ('ö', 'o'),
773            ('ç', 'c'),
774        ] {
775            expr = expr.replace(a, &b.to_string());
776        }
777        for w in [
778            "hesapla",
779            "kaç eder",
780            "kac eder",
781            "sonuç",
782            "sonuc",
783            "matematik",
784            "eşittir",
785            "esittir",
786        ] {
787            expr = expr.replace(w, " ");
788        }
789        expr = expr.replace("karekök", " ").replace("karekok", " ");
790        expr = expr.replace('×', "*").replace('÷', "/").replace(',', ".");
791        let mut s = String::new();
792        for c in expr.chars() {
793            if c.is_ascii_digit() || matches!(c, '+' | '-' | '*' | '/' | '.') {
794                s.push(c);
795            }
796        }
797        if s.is_empty() || !s.chars().any(|c| c.is_ascii_digit()) {
798            return None;
799        }
800        // çok basit sol-sağ değerlendirici (yalnızca * / sonra + -)
801        fn parse_num(tok: &str) -> Option<f64> {
802            tok.parse().ok()
803        }
804        let tokens: Vec<char> = s.chars().collect();
805        let mut nums = Vec::new();
806        let mut ops = Vec::new();
807        let mut cur = String::new();
808        for c in tokens {
809            if c.is_ascii_digit() || c == '.' {
810                cur.push(c);
811            } else if matches!(c, '+' | '-' | '*' | '/') {
812                if cur.is_empty() && c == '-' {
813                    cur.push('-');
814                    continue;
815                }
816                nums.push(parse_num(&cur)?);
817                cur.clear();
818                ops.push(c);
819            }
820        }
821        nums.push(parse_num(&cur)?);
822        // * /
823        let mut i = 0;
824        while i < ops.len() {
825            if ops[i] == '*' || ops[i] == '/' {
826                let a = nums[i];
827                let b = nums[i + 1];
828                let r = if ops[i] == '*' {
829                    a * b
830                } else if b.abs() < 1e-12 {
831                    return None;
832                } else {
833                    a / b
834                };
835                nums[i] = r;
836                nums.remove(i + 1);
837                ops.remove(i);
838            } else {
839                i += 1;
840            }
841        }
842        let mut acc = nums[0];
843        for (op, n) in ops.iter().zip(nums.iter().skip(1)) {
844            acc = match op {
845                '+' => acc + n,
846                '-' => acc - n,
847                _ => return None,
848            };
849        }
850        Some(acc)
851    }

853    /// Kullanıcı mesajını işle.
854    pub fn yanitla(&mut self, text: &str) -> Result<AlperYanit> {
855        let n = normalize_tr(text);
856        let mut classified = self.siniflandir(text)?;
857
858        // Kural güçlendirmeleri
859        if n.contains("benim adim ne") || n.contains("beni tani") {
860            classified.intent = "kullanici_hafiza".into();
861            classified.confidence = classified.confidence.max(0.9);
862        } else if n.contains("senin adin") || n == "adin ne" || n.contains("sen kimsin") {
863            classified.intent = "kimlik_sohbet".into();
864            classified.confidence = classified.confidence.max(0.9);
865        } else if (n.starts_with("neden")
866            || n.starts_with("detay")
867            || n.starts_with("peki")
868            || n.contains("bunu acikla")
869            || n.contains("ornek kod"))
870            && self.last_intent.is_some()
871            && classified.confidence < 0.85
872        {
873            classified.intent = "takip_detay".into();
874        }
875
876        let intent = classified.intent.as_str();
877        let conf = classified.confidence;
878        let greet = self
879            .profile
880            .name
881            .as_ref()
882            .map(|n| format!(", **{n}**"))
883            .unwrap_or_default();
884
885        let mut dsl = None;
886        let mut egitim_tetikle = None;
887        let metin = match intent {
888            "selamlama" => format!(
889                "Selam{greet}!\nBen **ALPER** (yerel Sequential, v{ALPER_VERSION}).\n- {} niyet · {:.1}% doğruluk{}\nNasıl yardımcı olayım?",
890                INTENTS.len(),
891                self.accuracy * 100.0,
892                if self.from_cache { " · önbellek" } else { "" }
893            ),
894            "hal_hatir" => format!(
895                "Teşekkürler{greet} — yerelde hazırım.\n- Beyin: %{:.1}\n- Profil: {}\nSizin gününüz nasıl?",
896                self.accuracy * 100.0,
897                self.profile.name.as_deref().unwrap_or("—")
898            ),
899            "kimlik_sohbet" => format!(
900                "**Ben ALPER.**\n- SphereX Labs × turk-ml-framework\n- TF-IDF+bigram → MLP 64→32→{}\n- Harici LLM yok\n- Önbellek: ~/.turk/alper_brain.bin",
901                INTENTS.len()
902            ),
903            "kullanici_hafiza" => {
904                if let Some(name) = Self::extract_name(text) {
905                    self.profile.name = Some(name.clone());
906                    let _ = self.profil_kaydet();
907                    format!("Memnun oldum, **{name}**!\nAdınız `~/.turk/alper_user_profile.json` dosyasına kaydedildi.")
908                } else if n.contains("adim ne") || n.contains("tani") || n.contains("kimim") {
909                    if let Some(ref name) = self.profile.name {
910                        format!("Sizi **{name}** olarak tanıyorum.")
911                    } else {
912                        "Henüz adınızı bilmiyorum. Örnek: «benim adım Sinan»".into()
913                    }
914                } else {
915                    format!(
916                        "Profil:\n- Ad: {}\n- Rol: {}",
917                        self.profile.name.as_deref().unwrap_or("(yok)"),
918                        self.profile.role.as_deref().unwrap_or("(yok)")
919                    )
920                }
921            }
922            "saat_tarih" => {
923                use std::time::SystemTime;
924                let _ = SystemTime::now();
925                // chrono feature olmayabilir — basit yerel
926                #[cfg(feature = "chrono")]
927                {
928                    let now = chrono::Local::now();
929                    format!(
930                        "**Yerel zaman**\n- {}",
931                        now.format("%A %d %B %Y · %H:%M:%S")
932                    )
933                }
934                #[cfg(not(feature = "chrono"))]
935                {
936                    "**Saat/tarih:** sistem saatine bakın (chrono özelliği kapalı). «hesapla» veya XOR eğitin.".into()
937                }
938            }
939            "hesaplama" => {
940                if let Some(v) = Self::safe_eval(text) {
941                    format!("**Hesap:** `{v}`")
942                } else {
943                    "Anlaşılır aritmetik bulamadım. Örnek: `hesapla 12*7+3`".into()
944                }
945            }
946            "model_egit_xor" => {
947                egitim_tetikle = Some("xor");
948                dsl = Some(xor_dsl().to_string());
949                format!("XOR eğitimini başlatıyorum{greet}. DSL editöre de yazıldı.")
950            }
951            "kod_uretme_dsl" => {
952                let deep = n.contains("derin") || n.contains("mlp") || n.contains("kume");
953                let code = if deep { derin_dsl() } else { xor_dsl() };
954                dsl = Some(code.to_string());
955                format!(
956                    "**Üretilen `.turk` DSL** ({})\nEditöre aktarıldı.",
957                    if deep { "derin" } else { "xor" }
958                )
959            }
960            "egitim_teshis" => {
961                "**Eğitim teşhisi**\n- Loss↓ yoksa: lr 0.01–0.05, batch küçült\n- Val↑ train↓: dropout / sabır\n- Acc~%50: etiket/özellik kontrol\n- XOR: 80+ epoch; derin: sabır=25".into()
962            }
963            "hiperparametre_ayar" => {
964                "**Hiperparametreler**\n- XOR: Adam lr=0.05, batch=4, ~120 epoch\n- Derin: lr=0.03, sabır=25\n- ALPER beyin: lr=0.035, batch=16, CE+Softmax".into()
965            }
966            "kavram_aciklama" => kavram_makale(text)
967                .unwrap_or(
968                    "**Kavram**\nReLU, Adam, Softmax, Dropout, WASM, Embedding, Conv1D deneyin.",
969                )
970                .to_string(),
971            "sistem_durumu" => format!(
972                "**Sistem**\n- ALPER v{ALPER_VERSION}\n- Doğruluk: %{:.1}\n- Örnek: {}\n- Vocab: {}\n- Önbellek: {}\n- Profil: {}",
973                self.accuracy * 100.0,
974                self.sample_count,
975                self.vocab.len(),
976                if self.from_cache { "evet" } else { "hayır (yeni eğitildi)" },
977                self.profile.name.as_deref().unwrap_or("—")
978            ),
979            "takip_detay" => {
980                let prev = self.last_topic.clone().unwrap_or_else(|| "genel".into());
981                if n.contains("uygula") || n.contains("egit") {
982                    egitim_tetikle = Some("xor");
983                    dsl = Some(xor_dsl().to_string());
984                    format!("Önceki konu `{prev}` → XOR eğitimi uygulanıyor.")
985                } else if let Some(art) = kavram_makale(&prev) {
986                    format!("**Takip (`{prev}`)**\n{art}")
987                } else {
988                    format!(
989                        "**Takip:** `{prev}`\nDaha spesifik sorun veya «örnek kod ver» / «uygula» deyin."
990                    )
991                }
992            }
993            _ => format!("Niyet `{intent}` işlendi (güven %{:.0}).", conf * 100.0),
994        };
995
996        self.last_intent = Some(intent.to_string());
997        self.last_topic = Some(if intent == "kavram_aciklama" {
998            normalize_tr(text)
999        } else {
1000            intent.to_string()
1001        });
1002
1003        Ok(AlperYanit {
1004            intent: intent.to_string(),
1005            confidence: conf,
1006            metin,
1007            dsl,
1008            egitim_tetikle,
1009        })
1010    }
