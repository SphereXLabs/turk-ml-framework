1//! Aktivasyon fonksiyonları modülü
2//!
3//! Bu modül, sinir ağı katmanlarında kullanılan aktivasyon fonksiyonlarını içerir.
4//! Her aktivasyon fonksiyonu için ileri yayılım (forward) ve geri yayılım (backward)
5//! hesaplamaları burada tanımlanmıştır.
6
7use crate::error::{Result, TurkError};
8use ndarray::{Array2, Axis};
9use serde::{Deserialize, Serialize};
10use std::fmt;
11
12use crate::dtype::Float;
13
14/// Aktivasyon fonksiyonları
15#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
16pub enum Activation {
17    Relu,
18    LeakyRelu(Float), // alpha değeri
19    Sigmoid,
20    Tanh,
21    Softmax,
22    Linear,
23    Elu(Float), // alpha değeri
24    Selu,
25    Swish,
26}
27
28impl Activation {
29    /// Metinden aktivasyon oluşturur
30    pub fn from_str(s: &str) -> Result<Self> {
31        match s.to_lowercase().as_str() {
32            "relu" => Ok(Self::Relu),
33            "leakyrelu" => Ok(Self::LeakyRelu(0.01)),
34            "sigmoid" => Ok(Self::Sigmoid),
35            "tanh" => Ok(Self::Tanh),
36            "softmax" => Ok(Self::Softmax),
37            "linear" => Ok(Self::Linear),
38            "elu" => Ok(Self::Elu(1.0)),
39            "selu" => Ok(Self::Selu),
40            "swish" => Ok(Self::Swish),
41            _ => Err(TurkError::LayerError(format!(
42                "Bilinmeyen aktivasyon: {}",
43                s
44            ))),
45        }
46    }
47
48    /// İleri hesaplama
49    pub fn forward(&self, x: &Array2<Float>) -> Array2<Float> {
50        match self {
51            Self::Relu => x.mapv(|v| if v > 0.0 { v } else { 0.0 }),
52            Self::LeakyRelu(alpha) => x.mapv(|v| if v > 0.0 { v } else { v * alpha }),
53            Self::Sigmoid => x.mapv(|v| 1.0 / (1.0 + (-v).exp())),
54            Self::Tanh => x.mapv(|v| v.tanh()),
55            Self::Softmax => {
56                let mut result = Array2::zeros(x.dim());
57                for (i, mut row) in result.axis_iter_mut(Axis(0)).enumerate() {
58                    let x_row = x.row(i);
59                    let max = x_row.fold(Float::NEG_INFINITY, |max, &v| max.max(v));
60                    let exp_x = x_row.mapv(|v| (v - max).exp());
61                    let sum_exp = exp_x.sum();
62                    for (j, v) in row.iter_mut().enumerate() {
63                        *v = exp_x[j] / sum_exp;
64                    }
65                }
66                result
67            }
68            Self::Linear => x.to_owned(),
69            Self::Elu(alpha) => x.mapv(|v| if v > 0.0 { v } else { alpha * (v.exp() - 1.0) }),
70            Self::Selu => x.mapv(|v| {
71                let alpha = crate::dtype::from_f64(1.673_263_242_354_377_2);
72                let scale = crate::dtype::from_f64(1.050_700_987_355_480_4);
73                if v > 0.0 {
74                    scale * v
75                } else {
76                    scale * alpha * (v.exp() - 1.0)
77                }
78            }),
79            Self::Swish => {
80                let sigmoid = x.mapv(|v| 1.0 / (1.0 + (-v).exp()));
81                sigmoid.clone() + x * &sigmoid.mapv(|v| v * (1.0 - v))
82            }
83        }
84    }
85
86    /// Geri hesaplama (türev)
87    pub fn backward(&self, x: &Array2<Float>) -> Array2<Float> {
88        match self {
89            Self::Relu => x.mapv(|v| if v > 0.0 { 1.0 } else { 0.0 }),
90            Self::LeakyRelu(alpha) => x.mapv(|v| if v > 0.0 { 1.0 } else { *alpha }),
91            Self::Sigmoid => {
92                let sigmoid = self.forward(x);
93                sigmoid.mapv(|v| v * (1.0 - v))
94            }
95            Self::Tanh => {
96                let tanh = x.mapv(|v| v.tanh());
97                tanh.mapv(|v| 1.0 - v * v)
98            }
99            Self::Softmax => {
100                // Softmax türevi karmaşıktır, genellikle cross-entropy ile birlikte kullanılır
101                // Bu durumda türev daha basittir (y_pred - y_true)
102                Array2::ones(x.dim())
103            }
104            Self::Linear => Array2::ones(x.dim()),
105            Self::Elu(alpha) => x.mapv(|v| if v > 0.0 { 1.0 } else { alpha * v.exp() }),
106            Self::Selu => x.mapv(|v| {
107                let alpha = crate::dtype::from_f64(1.673_263_242_354_377_2);
108                let scale = crate::dtype::from_f64(1.050_700_987_355_480_4);
109                if v > 0.0 {
110                    scale
111                } else {
112                    scale * alpha * v.exp()
113                }
114            }),
115            Self::Swish => {
116                let sigmoid = x.mapv(|v| 1.0 / (1.0 + (-v).exp()));
117                sigmoid.clone() + x * &sigmoid.mapv(|v| v * (1.0 - v))
118            }
119        }
120    }
121}

123impl fmt::Display for Activation {
124    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
125        match self {
126            Self::Relu => write!(f, "relu"),
127            Self::LeakyRelu(alpha) => write!(f, "leaky_relu(alpha={})", alpha),
128            Self::Sigmoid => write!(f, "sigmoid"),
129            Self::Tanh => write!(f, "tanh"),
130            Self::Softmax => write!(f, "softmax"),
131            Self::Linear => write!(f, "linear"),
132            Self::Elu(alpha) => write!(f, "elu(alpha={})", alpha),
133            Self::Selu => write!(f, "selu"),
134            Self::Swish => write!(f, "swish"),
135        }
136    }
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142    use ndarray::arr2;
143
144    #[test]
145    fn test_activation_functions() {
146        let input = arr2(&[[1.0, -1.0], [0.5, -0.5]]);
147
148        // ReLU testi
149        let relu_output = Activation::Relu.forward(&input);
150        assert_eq!(relu_output[[0, 0]], 1.0);
151        assert_eq!(relu_output[[0, 1]], 0.0);
152
153        // Sigmoid testi
154        let sigmoid_output = Activation::Sigmoid.forward(&input);
155        assert!(sigmoid_output[[0, 0]] > 0.5);
156        assert!(sigmoid_output[[0, 1]] < 0.5);
157
158        // Tanh testi
159        let tanh_output = Activation::Tanh.forward(&input);
160        assert!(tanh_output[[0, 0]] > 0.0);
161        assert!(tanh_output[[0, 1]] < 0.0);
162    }
163}
